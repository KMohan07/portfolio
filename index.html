def build_permission_filter(user):
    must_conditions = []
    if user.company:
        must_conditions.append(
            FieldCondition(
                key="company_id",
                match=MatchValue(value=str(user.company.id))
            )
        )

    should_conditions = [
        FieldCondition(
            key="allowed_roles",
            match=MatchAny(any=[user.role])
        )
    ]
    if user.department:
        should_conditions.append(
            FieldCondition(
                key="departments",
                match=MatchAny(any=[user.department])
            )
        )
    should_conditions.append(
        FieldCondition(
            key="specific_users",
            match=MatchAny(any=[str(user.id)])
        )
    )
    return Filter(
        must=must_conditions,
        should=should_conditions
    )

def process_document(document_id):
    document = Document.objects.get(id=document_id)
    document.status = "processing"
    document.save()
    
    try:
        # Load file content depending on type
        text_content = parse_file(document.file.path, document.document_type)
        
        # Extract chunks and build local nodes using LlamaIndex
        chunks = chunk_text(text_content)
        
        # Push vectors to Qdrant collection with security payloads
        index_chunks_in_qdrant(document, chunks)
        
        document.status = "indexed"
        document.save()
    except Exception as e:
        document.status = "failed"
        document.error_message = str(e)
        document.save()

def is_memory_question(question):
    question = question.lower()
    memory_keywords = [
        "previous question",
        "previous message",
        "what did i ask",
        "summarize our chat",
        "summarize conversation",
        "chat summary",
    ]
    return any(kw in question for kw in memory_keywords)

def is_greeting_or_general(question):
    q = question.strip().lower().rstrip("?.!")
    greetings = {"hi", "hello", "hey", "who are you", "help"}
    if q in greetings: return True
    return False
